		\NewDocumentCommand{\delimAux}{somm}{
			\IfBooleanTF{#1}{%
				\csname#3\endcsname*{#4}%
			}{\IfNoValueTF{#2}{%
					\csname#3\endcsname{#4}%
				}{%
					\csname#3\endcsname[#2]{#4}%
				}%
			}%
		}
		\NewDocumentCommand{\delimAuxSub}{sommO{}}{
			\IfBooleanTF{#1}

		}
